cluster, retrieve the VPN URL with:
  PROMETHEUS_VPN_IP=$(kubectl -n {{ .Release.Namespace }} get pod \
    -l app.kubernetes.io/component=prometheus \
    -o jsonpath='{.items[0].status.hostIP}')
  echo "http://${PROMETHEUS_VPN_IP}:{{ .Values.monitoring.prometheus.service.port }}"
